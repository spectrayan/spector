1. Get the application URL by running these commands:
{{- if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "spector.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo "Spector Dashboard: http://$NODE_IP:$NODE_PORT"
{{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "spector.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "spector.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo "Spector Dashboard: http://$SERVICE_IP:{{ .Values.service.ports.dashboard }}"
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "{{ include "spector.selectorLabels" . }}" -o jsonpath="{.items[0].metadata.name}")
  echo "Forwarding Dashboard (8080 -> 7700) and API (7070 -> 7070)..."
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 7700:8080 7070:7070 &
  echo "Spector Dashboard: http://127.0.0.1:7700"
  echo "Spector API/MCP:   http://127.0.0.1:7070"
  echo "Health Endpoint:   http://127.0.0.1:7070/actuator/health"
{{- end }}
